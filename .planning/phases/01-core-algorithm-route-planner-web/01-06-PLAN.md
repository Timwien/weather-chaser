    - At least one criterion must remain selected at all times — if user tries to deselect the last one, keep it selected

    --- Update apps/web/src/components/map/MapContainer.tsx ---
    `DrawingControls` uses `useMap()` from `@vis.gl/react-maplibre` and MUST be rendered as a
    child inside the `<Map>` element (not as a sibling outside it). Add two optional props to
    `MapContainer` so the entry panel can trigger polygon drawing without needing to reach
    inside the map:

    ```tsx
    interface MapContainerProps {
      onDrawComplete?: (polygon: [number, number][]) => void;
      onDrawClear?: () => void;
    }

    export function MapContainer({ onDrawComplete, onDrawClear }: MapContainerProps) {
      return (
        <div className="map-root">
          <Map
            initialViewState={{ longitude: 10.4515, latitude: 51.1657, zoom: 5.5 }}
            mapStyle={MAP_STYLE}
            style={{ width: '100%', height: '100%' }}
          >
            {/* DrawingControls must live inside <Map> so useMap() has a provider */}
            {onDrawComplete && onDrawClear && (
              <DrawingControls
                onPolygonComplete={onDrawComplete}
                onClear={onDrawClear}
              />
            )}
          </Map>
        </div>
      );
    }
    ```
