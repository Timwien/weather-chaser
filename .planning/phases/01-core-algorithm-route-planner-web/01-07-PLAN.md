    --- apps/web/src/services/weather.ts ---
    Fetch hourly weather for a list of towns from Open-Meteo. Returns data in a form that scoreLocation can consume.

    ```typescript
    import type { Town, HourlyWeather } from '@weatherchaser/core';

    export interface WeatherData {
      townId: string;
      hourly: HourlyWeather;
    }

    // Open-Meteo supports comma-separated lat/lng arrays for multi-location fetch
    // See RESEARCH.md Pattern 6
    export async function fetchWeatherBatch(
      towns: Town[],
      startDate: string,    // YYYY-MM-DD
      endDate: string,      // YYYY-MM-DD
    ): Promise<WeatherData[]> {
      if (towns.length === 0) return [];

      // Open-Meteo supports arrays via repeated params or comma-separated
      // Batch in groups of 50 if more than 50 towns (safety limit)
      const BATCH_SIZE = 50;
      const results: WeatherData[] = [];

      for (let i = 0; i < towns.length; i += BATCH_SIZE) {
        const batch = towns.slice(i, i + BATCH_SIZE);
        const batchResults = await fetchBatch(batch, startDate, endDate);
        results.push(...batchResults);
      }

      return results;
    }

    async function fetchBatch(
      towns: Town[],
      startDate: string,
      endDate: string,
    ): Promise<WeatherData[]> {
      const url = new URL('https://api.open-meteo.com/v1/forecast');
      url.searchParams.set('latitude', towns.map((t) => t.lat.toFixed(4)).join(','));
      url.searchParams.set('longitude', towns.map((t) => t.lng.toFixed(4)).join(','));
      url.searchParams.set('hourly', 'temperature_2m,precipitation,sunshine_duration,wind_speed_10m');
      url.searchParams.set('start_date', startDate);
      url.searchParams.set('end_date', endDate);
      url.searchParams.set('timezone', 'auto');

      const res = await fetch(url.toString());
      if (!res.ok) throw new Error(`Open-Meteo error: ${res.status}`);

      // When multiple lat/lng: response is an array. When single: response is an object.
      const raw = await res.json();
      const dataArray: Array<{ hourly: HourlyWeather }> = Array.isArray(raw) ? raw : [raw];

      return towns.map((town, idx) => ({
        townId: town.id,
        hourly: dataArray[idx]?.hourly ?? { time: [], temperature_2m: [], precipitation: [], sunshine_duration: [], wind_speed_10m: [] },
      }));
    }
    ```

    --- apps/web/src/services/osrm.ts ---
    Pre-compute the full NxN distance matrix using OSRM /table endpoint.
    Phase 1 dev: points to a locally-running OSRM Docker instance at http://localhost:5000.
    Use an environment variable VITE_OSRM_URL so it can be pointed at a proxy later.

    ```typescript
    import type { Town } from '@weatherchaser/core';

    // Default: local OSRM Docker in development
    // Set VITE_OSRM_URL in .env.local to override
    const OSRM_URL = import.meta.env.VITE_OSRM_URL ?? 'http://localhost:5000';

    export interface DistanceMatrix {
      durations: number[][];   // seconds NxN
      distances: number[][];   // meters NxN
    }

    // OSRM /table/v1 endpoint — one call returns full NxN matrix
    // ALGO-06: pre-computed once per session, not called per optimizer step
    export async function fetchDistanceMatrix(towns: Town[]): Promise<DistanceMatrix> {
      if (towns.length === 0) return { durations: [], distances: [] };

      // OSRM format: lng,lat;lng,lat;...
      const coords = towns.map((t) => `${t.lng},${t.lat}`).join(';');
      const url = `${OSRM_URL}/table/v1/driving/${coords}?annotations=duration,distance`;

      const res = await fetch(url);
      if (!res.ok) throw new Error(`OSRM error: ${res.status} — is OSRM running at ${OSRM_URL}?`);

      const data: { durations: number[][]; distances: number[][] } = await res.json();
      return { durations: data.durations, distances: data.distances };
    }
    ```

    The worker is the orchestration core. It receives an OptimizerWorkerInput message, runs the full pipeline, and posts back progress events and a final result.

    ```typescript
    // apps/web/src/workers/optimizer.worker.ts
    // Vite module worker (type: 'module') — can import from packages/core

    import { optimizeRoute, scoreLocation, PRESETS } from '@weatherchaser/core';
    import type { Town, OptimizerInput, WeatherScore, ScoringWeights } from '@weatherchaser/core';
    import { fetchWeatherBatch } from '../services/weather.ts';
    import { fetchDistanceMatrix } from '../services/osrm.ts';
    import { fetchTownsInArea, fetchTownsInPolygon } from '../services/overpass.ts';
    import type { BoundingBox } from '../services/nominatim.ts';

    export interface OptimizerWorkerInput {
      type: 'run';
      searchArea: {
        type: 'place' | 'polygon' | 'radius';
        bbox?: [number, number, number, number]; // [west, south, east, north]
        polygon?: [number, number][];
      };
      config: {
        startDate: string;       // YYYY-MM-DD
        endDate: string;         // YYYY-MM-DD
        totalDays: number;
        maxStay: number;
        preset: string;
        startLat: number | null;
        startLng: number | null;
        mustVisitCoords: Array<{ lat: number; lng: number; name: string }>;
      };
    }

    export type OptimizerWorkerOutput =
      | { type: 'progress'; step: 'finding_towns' | 'fetching_weather' | 'optimizing_route' }
      | { type: 'complete'; result: ReturnType<typeof optimizeRoute> }
      | { type: 'error'; message: string };

    self.onmessage = async (event: MessageEvent<OptimizerWorkerInput>) => {
      if (event.data.type !== 'run') return;
      const { searchArea, config } = event.data;

      try {
        // Step 1: Fetch towns
        self.postMessage({ type: 'progress', step: 'finding_towns' } satisfies OptimizerWorkerOutput);

        let towns: Town[];
        if (searchArea.type === 'polygon' && searchArea.polygon) {
          towns = await fetchTownsInPolygon(searchArea.polygon);
        } else if (searchArea.bbox) {
          const [west, south, east, north] = searchArea.bbox;
          towns = await fetchTownsInArea({ south, north, west, east } as BoundingBox);
        } else {
          throw new Error('No search area defined');
        }

        if (towns.length === 0) {
          self.postMessage({ type: 'error', message: 'no_towns' } satisfies OptimizerWorkerOutput);
          return;
        }

        // Inject must-visit towns if not already in the list (by proximity check)
        const mustVisitTowns: Town[] = config.mustVisitCoords.map((c, i) => ({
          id: `must-visit-${i}`,
          name: c.name,
          lat: c.lat,
          lng: c.lng,
        }));
        // Add must-visit towns that aren't already nearby in the towns list (simple name check)
        for (const mv of mustVisitTowns) {
          const alreadyPresent = towns.some(
            (t) => t.name.toLowerCase() === mv.name.toLowerCase()
          );
          if (!alreadyPresent) towns.push(mv);
        }

        // Determine start index
        let startIndex = 0;
        if (config.startLat !== null && config.startLng !== null) {
          const startLat = config.startLat;
          const startLng = config.startLng;
          let minDist = Infinity;
          towns.forEach((t, i) => {
            const d = Math.hypot(t.lat - startLat, t.lng - startLng);
            if (d < minDist) { minDist = d; startIndex = i; }
          });
        }

        // Determine must-visit indices
        const mustVisitIndices = mustVisitTowns.map((mv) =>
          towns.findIndex((t) => t.name.toLowerCase() === mv.name.toLowerCase())
        ).filter((i) => i !== -1);

        // Step 2: Fetch weather
        self.postMessage({ type: 'progress', step: 'fetching_weather' } satisfies OptimizerWorkerOutput);
        const weatherData = await fetchWeatherBatch(towns, config.startDate, config.endDate);

        // Pre-score each town for its estimated arrival date
        // For pre-scoring: use a simple estimate (arrival day 1 for first stop, etc.)
        // Full temporal scoring happens in assignStops inside optimizeRoute
        // Here we pass per-town weather data — assignStops calls scoreLocation with actual arrival dates
        const weights: ScoringWeights = {
          ...(PRESETS[config.preset as keyof typeof PRESETS] ?? PRESETS['sightseeing']),
        };

        const weatherScores: WeatherScore[] = towns.map((town, i) => {
          const data = weatherData.find((d) => d.townId === town.id);
          if (!data) {
            return { composite: 0, breakdown: { sunshine: 0, precipitation: 0, temperature: 0, wind: 0 } };
          }
          // Initial score using startDate — assignStops re-scores with actual arrival dates per stop
          return scoreLocation(data.hourly, new Date(config.startDate), 1, weights);
        });

        // Step 3: Fetch distance matrix and optimize
        self.postMessage({ type: 'progress', step: 'optimizing_route' } satisfies OptimizerWorkerOutput);

        const matrix = await fetchDistanceMatrix(towns);

        // Convert distances from meters to km for the optimizer
        const distanceKm = matrix.distances.map((row) => row.map((d) => d / 1000));

        const optimizerInput: OptimizerInput = {
          towns,
          distanceMatrix: distanceKm,
          durationMatrix: matrix.durations,
          weatherScores,
          config: {
            startIndex,
            totalDays: config.totalDays,
            maxStay: config.maxStay,
            mustVisitIndices,
            startDate: new Date(config.startDate),
            weights,
          },
        };

        const result = optimizeRoute(optimizerInput);
        self.postMessage({ type: 'complete', result } satisfies OptimizerWorkerOutput);
      } catch (err) {
        self.postMessage({
          type: 'error',
          message: err instanceof Error ? err.message : 'unknown_error',
        } satisfies OptimizerWorkerOutput);
      }
    };
    ```

    Add a Haversine fallback in osrm.ts:
    ```typescript
    function haversineKm(lat1: number, lng1: number, lat2: number, lng2: number): number {
      const R = 6371;
      const dLat = ((lat2 - lat1) * Math.PI) / 180;
      const dLng = ((lng2 - lng1) * Math.PI) / 180;
      const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLng/2)**2;
      return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    }

    export async function fetchDistanceMatrix(towns: Town[]): Promise<DistanceMatrix> {
      if (towns.length === 0) return { durations: [], distances: [] };
      try {
        // ... existing OSRM fetch code ...
      } catch {
        // OSRM not available — fall back to straight-line distances
        console.warn('OSRM unavailable, using straight-line distances');
        const n = towns.length;
        const distances = Array.from({ length: n }, (_, i) =>
          Array.from({ length: n }, (_, j) =>
            haversineKm(towns[i].lat, towns[i].lng, towns[j].lat, towns[j].lng) * 1000 // meters
          )
        );
        const durations = distances.map(row => row.map(d => (d / 1000) * 60)); // rough 60km/h estimate
        return { distances, durations };
      }
    }
    ```

    --- apps/web/src/hooks/useOptimizer.ts ---
    React hook that manages the Web Worker lifecycle:

    ```typescript
    import { useCallback, useEffect, useRef } from 'react';
    import { useAppStore } from '../stores/appStore.ts';
    import type { OptimizerWorkerInput, OptimizerWorkerOutput } from '../workers/optimizer.worker.ts';

    export function useOptimizer() {
      const workerRef = useRef<Worker | null>(null);
      const { searchArea, tripConfig, setMode, setLoadingStep, setRoute, setError } = useAppStore();

      useEffect(() => {
        // Clean up worker on unmount
        return () => {
          workerRef.current?.terminate();
          workerRef.current = null;
        };
      }, []);

      const run = useCallback(() => {
        if (!searchArea || !tripConfig.startDate || !tripConfig.endDate) return;

        // Terminate any existing worker
        workerRef.current?.terminate();

        // Create new worker — Vite resolves the URL at build time
        workerRef.current = new Worker(
          new URL('../workers/optimizer.worker.ts', import.meta.url),
          { type: 'module' },
        );

        setMode('loading');
        setLoadingStep('finding_towns');
        setError(null);
        setRoute(null);

        workerRef.current.onmessage = (event: MessageEvent<OptimizerWorkerOutput>) => {
          const msg = event.data;
          if (msg.type === 'progress') {
            setLoadingStep(msg.step);
          } else if (msg.type === 'complete') {
            setRoute(msg.result);
            setMode('results');
            setLoadingStep(null);
          } else if (msg.type === 'error') {
            setError(msg.message);
            setMode('idle');
            setLoadingStep(null);
          }
        };

        workerRef.current.onerror = (err) => {
          setError(err.message);
          setMode('idle');
          setLoadingStep(null);
        };

        const input: OptimizerWorkerInput = {
          type: 'run',
          searchArea: {
            type: searchArea.type,
            bbox: searchArea.bbox ? [
              searchArea.bbox[0], searchArea.bbox[1],
              searchArea.bbox[2], searchArea.bbox[3],
            ] : undefined,
            polygon: searchArea.polygon as [number, number][] | undefined,
          },
          config: {
            startDate: tripConfig.startDate,
            endDate: tripConfig.endDate,
            totalDays: tripConfig.totalDays,
            maxStay: tripConfig.maxStay,
            preset: tripConfig.preset,
            startLat: tripConfig.startLat,
            startLng: tripConfig.startLng,
            mustVisitCoords: tripConfig.mustVisitCoords,
          },
        };

        workerRef.current.postMessage(input);
      }, [searchArea, tripConfig, setMode, setLoadingStep, setRoute, setError]);

      return { run };
    }
    ```

    ```tsx
    import { useTranslation } from 'react-i18next';
    import { useAppStore } from '../../stores/appStore.ts';
    import './LoadingOverlay.css';

    const STEPS = ['finding_towns', 'fetching_weather', 'optimizing_route'] as const;

    export function LoadingOverlay() {
      const { t } = useTranslation('common');
      const { mode, loadingStep } = useAppStore();

      if (mode !== 'loading') return null;

      return (
        <div className="loading-overlay">
          <div className="loading-card">
            <div className="loading-spinner" aria-hidden="true" />
            <div className="loading-steps">
              {STEPS.map((step) => {
                const currentIdx = STEPS.indexOf(loadingStep ?? 'finding_towns');
                const stepIdx = STEPS.indexOf(step);
                const isDone = stepIdx < currentIdx;
                const isCurrent = step === loadingStep;
                return (
                  <div
                    key={step}
                    className={`loading-step ${isDone ? 'done' : ''} ${isCurrent ? 'current' : ''}`}
                  >
                    <span className="loading-step-indicator">
                      {isDone ? '✓' : isCurrent ? '›' : '·'}
                    </span>
                    <span>{t(`loading.${step}`)}</span>
                  </div>
                );
              })}
            </div>
          </div>
        </div>
      );
    }
    ```
